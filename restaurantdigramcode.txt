
Table Users {
  user_id int [pk]
  Fullname varchar(250)
  phone varchar(50) [unique]
  email varchar(100) [unique]
  address varchar(250)
  passwordhash varchar(255)
  role enum(250)
  created_at timestamp
}

Table Restaurants {
  restaurant_id int [pk]
  owner_id int [ref: > Users.user_id]
  Name varchar(50)
  phone varchar(50) [unique]
  address varchar(250)
  opentime timestamp
  closingtime timestamp
  Description varchar(250)
}

Table Categories {
  categories_id int [pk]
  Name varchar(100)
  restaurant_id int [ref: > Restaurants.restaurant_id]
}



Table Products {
  product_id int [pk]
  Name varchar(100)
  Price decimal
  categories_id int [ref: > Categories.categories_id]
  is_available bool
  Description varchar(250)
}


Table Orders {
  order_id int [pk]
  CustomerId int [ref: > Users.user_id]
  RestaurantId int [ref: > Restaurants.restaurant_id]
  status_name enum(50)
  OrderDate datetime
  TotalPrice decimal
}

Table OrderItems {
  orderItems_id int [pk]
  order_id int [ref: > Orders.order_id]
  Product_id int [ref: > Products.product_id]
  Quantity int
  Price decimal
}
